cmake_minimum_required(VERSION 3.21)
project(MusiKCpp VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Multimedia OpenGLWidgets OpenGL Concurrent)
qt_add_executable(MusiKCpp src/main.cpp src/circler_scene.cpp)
target_link_libraries(MusiKCpp PRIVATE Qt6::Widgets Qt6::Multimedia Qt6::OpenGLWidgets Qt6::OpenGL Qt6::Concurrent)
option(MUSIK_BUILD_TESTS "Build native borderless window regression" OFF)
if(MUSIK_BUILD_TESTS)
    enable_testing()
    qt_add_executable(borderless_regression tests/borderless.cpp)
    add_dependencies(borderless_regression MusiKCpp)
    target_link_libraries(borderless_regression PRIVATE Qt6::Widgets Qt6::Multimedia Qt6::OpenGLWidgets Qt6::OpenGL Qt6::Concurrent)
    add_test(NAME borderless_regression COMMAND borderless_regression)
    qt_add_executable(player_regression tests/player.cpp)
    add_dependencies(player_regression MusiKCpp)
    target_link_libraries(player_regression PRIVATE Qt6::Widgets Qt6::Multimedia Qt6::OpenGLWidgets Qt6::OpenGL Qt6::Concurrent)
    add_test(NAME player_regression COMMAND player_regression)
    qt_add_executable(circler_regression tests/circler.cpp src/circler_scene.cpp)
    target_link_libraries(circler_regression PRIVATE Qt6::Widgets Qt6::Multimedia Qt6::OpenGLWidgets Qt6::OpenGL Qt6::Concurrent)
    add_test(NAME circler_regression COMMAND circler_regression)
    qt_add_executable(seek_regression tests/seek.cpp)
    target_link_libraries(seek_regression PRIVATE Qt6::Widgets)
    add_test(NAME seek_regression COMMAND seek_regression)
endif()
add_custom_command(TARGET MusiKCpp POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/shaders" "$<TARGET_FILE_DIR:MusiKCpp>/shaders")
add_custom_command(TARGET MusiKCpp POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/MusiK-logo.png" "$<TARGET_FILE_DIR:MusiKCpp>/MusiK-logo.png")
