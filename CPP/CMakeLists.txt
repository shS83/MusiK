cmake_minimum_required(VERSION 3.21)
project(MusiKCpp VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Multimedia OpenGLWidgets OpenGL)
qt_add_executable(MusiKCpp src/main.cpp)
target_link_libraries(MusiKCpp PRIVATE Qt6::Widgets Qt6::Multimedia Qt6::OpenGLWidgets Qt6::OpenGL)
option(MUSIK_BUILD_TESTS "Build native borderless window regression" OFF)
if(MUSIK_BUILD_TESTS)
    enable_testing()
    qt_add_executable(borderless_regression tests/borderless.cpp)
    add_dependencies(borderless_regression MusiKCpp)
    target_link_libraries(borderless_regression PRIVATE Qt6::Widgets Qt6::Multimedia Qt6::OpenGLWidgets Qt6::OpenGL)
    add_test(NAME borderless_regression COMMAND borderless_regression)
endif()
add_custom_command(TARGET MusiKCpp POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/shaders" "$<TARGET_FILE_DIR:MusiKCpp>/shaders")
add_custom_command(TARGET MusiKCpp POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/MusiK-logo.png" "$<TARGET_FILE_DIR:MusiKCpp>/MusiK-logo.png")
